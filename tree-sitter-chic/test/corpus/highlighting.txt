/// A counter used by the demo.
Counter : struct {
    value : i32 = 0
}

@extension
increment : func(self : ^Counter, amount : i32 = 1) {
    self.value += amount
}

main : func() -> i32 {
    var counter := Counter {}
    text := "Chic\n"
    c_text : cstring = c"hello"
    rune_value := r'λ'
    mask := 0xff

    for i in 0..10 step 2 {
        if i >= 4 && true {
            counter.increment(i)
        }
    }

    return cast<i32>(counter.value)
}
